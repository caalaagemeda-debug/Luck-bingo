// server.js (Node.js backend snippet) const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const app = express(); const server = http.createServer(app); const io = new Server(server, { cors: { origin: "*" } }); let gameState = { prizePool: 0, timer: 50, drawnBalls: [], inProgress: false, players: {} }; // Game Loop: Pre-game countdown & Ball drawing setInterval(() => { if (!gameState.inProgress) { if (gameState.timer > 0) { gameState.timer--; io.emit('timerUpdate', gameState.timer); } else { gameState.inProgress = true; gameState.timer = 0; io.emit('gameStarted'); startBallDrawing(); } } }, 1000); function startBallDrawing() { let interval = setInterval(() => { if (gameState.drawnBalls.length < 75 && gameState.inProgress) { let nextBall; do { nextBall = Math.floor(Math.random() * 75) + 1; } while (gameState.drawnBalls.includes(nextBall)); gameState.drawnBalls.push(nextBall); io.emit('newBall', nextBall); } else { clearInterval(interval); } }, 2000); } io.on('connection', (socket) => { socket.emit('syncState', gameState); socket.on('buyTicket', (data) => { if (!gameState.inProgress) { gameState.prizePool += data.ticketPrice * 0.8; // 80% to pool, 20% house fee io.emit('prizePoolUpdate', gameState.prizePool); } }); socket.on('claimBingo', (ticketData) => { // Validate ticket on server side against gameState.drawnBalls const isValidWin = validateTicket(ticketData, gameState.drawnBalls); if (isValidWin && gameState.inProgress) { gameState.inProgress = false; io.emit('gameOver', { winner: socket.id, prize: gameState.prizePool }); // Reset for next round setTimeout(resetGame, 5000); } }); }); function validateTicket(ticket, drawn) { // Server-side verification logic for 5x5 grid patterns return true; } function resetGame() { gameState = { prizePool: 0, timer: 50, drawnBalls: [], inProgress: false, players: {} }; io.emit('syncState', gameState); } server.listen(3000, () => console.log('Bingo server running on port 3000'));
