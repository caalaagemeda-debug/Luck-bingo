// State Management const gameState = { activeScreen: 'lobby', // 'lobby' or 'game' selectedTicket: 97, ticketPrice: 10, balance: 5.00, jackpot: 1024, timerSeconds: 17, takenNumbers: [4, 6, 7, 8, 9, 13, 14, 15, 16, 22, 42, 43, 44, 59, 60, 62, 64, 67, 68, 72, 73, 74, 80, 81, 84, 88, 90, 92, 99], // Game round state currentBall: 'B4', calledBalls: ['B4', 'I29', 'B12', 'B5'], // Player's 5x5 card layout [B, I, N, G, O] cardNumbers: [ [2, 13, 10, 4, 5], // B column [28, 21, 25, 30, 23], // I column [31, 43, '★', 34, 41], // N column (Center FREE) [51, 53, 57, 49, 46], // G column [75, 74, 68, 71, 69] // O column ], markedPositions: new Set(['2-2', '0-3', '0-4']) // 'N-center', 'B-4', 'B-5' }; // 1. Generate the 1-100 Ticket Selection Grid function initTicketGrid(gridElementId) { const container = document.getElementById(gridElementId); if (!container) return; container.innerHTML = ''; for (let i = 1; i <= 100; i++) { const btn = document.createElement('div'); btn.className = 'num-btn'; btn.textContent = i; if (gameState.takenNumbers.includes(i)) { btn.classList.add('taken'); } else if (i === gameState.selectedTicket) { btn.classList.add('active'); } btn.onclick = () => selectTicket(i, btn); container.appendChild(btn); } } // 2. Select / Deselect a Card function selectTicket(ticketNumber, element) { if (gameState.takenNumbers.includes(ticketNumber)) return; gameState.selectedTicket = ticketNumber; // Update UI active state document.querySelectorAll('.num-btn').forEach(btn => btn.classList.remove('active')); element.classList.add('active'); updateLobbyUI(); } // 3. Screen Switcher function switchView(targetScreen) { const lobbyScreen = document.getElementById('screen-lobby'); const gameScreen = document.getElementById('screen-game'); gameState.activeScreen = targetScreen || (gameState.activeScreen === 'lobby' ? 'game' : 'lobby'); if (gameState.activeScreen === 'game') { lobbyScreen.style.display = 'none'; gameScreen.style.display = 'flex'; } else { lobbyScreen.style.display = 'flex'; gameScreen.style.display = 'none'; } } // 4. Pre-Game Timer System function startLobbyTimer(timerElementId) { const timerElem = document.getElementById(timerElementId); setInterval(() => { if (gameState.timerSeconds > 0) { gameState.timerSeconds--; } else { gameState.timerSeconds = 30; // Reset countdown loop } if (timerElem) { const secs = gameState.timerSeconds; timerElem.textContent = `00:${secs < 10 ? '0' : ''}${secs}`; } }, 1000); } // 5. Drawing and Marking Bingo Balls function drawNextBall() { const letters = ['B', 'I', 'N', 'G', 'O']; const letter = letters[Math.floor(Math.random() * letters.length)]; let number; switch (letter) { case 'B': number = Math.floor(Math.random() * 15) + 1; break; case 'I': number = Math.floor(Math.random() * 15) + 16; break; case 'N': number = Math.floor(Math.random() * 15) + 31; break; case 'G': number = Math.floor(Math.random() * 15) + 46; break; case 'O': number = Math.floor(Math.random() * 15) + 61; break; } const ballCode = `${letter}${number}`; gameState.currentBall = ballCode; gameState.calledBalls.unshift(ballCode); updateGameplayUI(); } // 6. Check for Winning Line (Bingo) function checkBingo() { const grid = gameState.markedPositions; // Check Columns & Rows for (let i = 0; i < 5; i++) { let colComplete = true; let rowComplete = true; for (let j = 0; j < 5; j++) { if (!grid.has(`${i}-${j}`)) colComplete = false; if (!grid.has(`${j}-${i}`)) rowComplete = false; } if (colComplete || rowComplete) return true; } // Check Diagonals let diag1 = true; let diag2 = true; for (let i = 0; i < 5; i++) { if (!grid.has(`${i}-${i}`)) diag1 = false; if (!grid.has(`${i}-${4 - i}`)) diag2 = false; } return diag1 || diag2; } // UI Sync Helpers function updateLobbyUI() { const countElem = document.getElementById('selected-count'); const betElem = document.getElementById('total-bet-val'); if (countElem) countElem.textContent = gameState.selectedTicket ? 1 : 0; if (betElem) betElem.textContent = `${gameState.selectedTicket ? gameState.ticketPrice : 0} ETB`; } function updateGameplayUI() { const currentBallElem = document.getElementById('current-ball-num'); const historyElem = document.getElementById('ball-history-list'); if (currentBallElem) { currentBallElem.textContent = gameState.currentBall.replace(/^[A-Z]/, ''); } if (historyElem) { historyElem.innerHTML = gameState.calledBalls .map(ball => `
${ball}
`) .join(''); } } // Initialize scripts on page load document.addEventListener('DOMContentLoaded', () => { initTicketGrid('ticket-grid'); startLobbyTimer('lobby-timer'); });
